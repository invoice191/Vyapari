### SECTION A1 — RFM Customer Segmentation

```
Build a complete RFM (Recency, Frequency, Monetary) segmentation system for Vyapari.

─── POSTGRES FUNCTION ─────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION calculate_rfm(p_business_id UUID)
RETURNS TABLE (
  contact_id UUID,
  contact_name TEXT,
  recency_days INT,
  frequency INT,
  monetary NUMERIC,
  r_score INT,
  f_score INT,
  m_score INT,
  rfm_segment TEXT,
  segment_label TEXT,
  segment_color TEXT
) AS $$
BEGIN
  RETURN QUERY
  WITH base AS (
    SELECT
      c.id AS contact_id,
      c.name AS contact_name,
      EXTRACT(DAY FROM NOW() - MAX(i.created_at))::INT AS recency_days,
      COUNT(i.id)::INT AS frequency,
      COALESCE(SUM(i.total_amount), 0) AS monetary
    FROM contacts c
    LEFT JOIN invoices i ON i.contact_id = c.id
      AND i.business_id = p_business_id
      AND i.type = 'sale'
    WHERE c.business_id = p_business_id
      AND c.type = 'customer'
    GROUP BY c.id, c.name
  ),
  scored AS (
    SELECT *,
      NTILE(5) OVER (ORDER BY recency_days ASC)  AS r_score,
      NTILE(5) OVER (ORDER BY frequency DESC)     AS f_score,
      NTILE(5) OVER (ORDER BY monetary DESC)      AS m_score
    FROM base
  )
  SELECT
    s.contact_id, s.contact_name,
    s.recency_days, s.frequency, s.monetary,
    s.r_score, s.f_score, s.m_score,
    CONCAT(s.r_score, s.f_score, s.m_score) AS rfm_segment,
    CASE
      WHEN s.r_score >= 4 AND s.f_score >= 4 AND s.m_score >= 4 THEN 'Champion'
      WHEN s.r_score >= 3 AND s.f_score >= 3 THEN 'Loyal'
      WHEN s.r_score >= 4 AND s.f_score <= 2 THEN 'New Customer'
      WHEN s.r_score >= 3 AND s.f_score <= 2 AND s.m_score >= 3 THEN 'Potential Loyalist'
      WHEN s.r_score <= 2 AND s.f_score >= 3 AND s.m_score >= 3 THEN 'At Risk'
      WHEN s.r_score <= 2 AND s.f_score >= 4 THEN 'Cannot Lose'
      WHEN s.r_score <= 1 AND s.f_score <= 2 THEN 'Lost'
      ELSE 'Needs Attention'
    END AS segment_label,
    CASE
      WHEN s.r_score >= 4 AND s.f_score >= 4 AND s.m_score >= 4 THEN '#FF5500'
      WHEN s.r_score <= 2 AND s.f_score >= 3 THEN '#FF0000'
      WHEN s.r_score <= 1 THEN '#888888'
      ELSE '#FFA500'
    END AS segment_color
  FROM scored s;
END;
$$ LANGUAGE plpgsql;

-- Store results for fast UI access (refresh nightly via pg_cron)
CREATE TABLE IF NOT EXISTS rfm_results (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  business_id UUID NOT NULL,
  contact_id UUID NOT NULL,
  contact_name TEXT,
  recency_days INT,
  frequency INT,
  monetary NUMERIC,
  r_score INT, f_score INT, m_score INT,
  rfm_segment TEXT,
  segment_label TEXT,
  segment_color TEXT,
  calculated_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(business_id, contact_id)
);

-- pg_cron job: refresh every night at 2am
SELECT cron.schedule('rfm-refresh', '0 2 * * *',
  $$INSERT INTO rfm_results
    SELECT gen_random_uuid(), business_id, contact_id, contact_name,
           recency_days, frequency, monetary, r_score, f_score, m_score,
           rfm_segment, segment_label, segment_color, NOW()
    FROM businesses b
    CROSS JOIN LATERAL calculate_rfm(b.id)
    ON CONFLICT (business_id, contact_id) DO UPDATE SET
      recency_days = EXCLUDED.recency_days,
      frequency = EXCLUDED.frequency,
      monetary = EXCLUDED.monetary,
      rfm_segment = EXCLUDED.rfm_segment,
      segment_label = EXCLUDED.segment_label,
      calculated_at = NOW()$$
);

import { useEffect, useState } from 'react'
import { supabase } from '@/lib/supabase'

export type RFMSegment = 'Champion' | 'Loyal' | 'At Risk' | 'Cannot Lose' |
                          'New Customer' | 'Potential Loyalist' | 'Lost' | 'Needs Attention'

export interface RFMResult {
  contact_id: string
  contact_name: string
  recency_days: number
  frequency: number
  monetary: number
  rfm_segment: string
  segment_label: RFMSegment
  segment_color: string
}

export function useRFM(businessId: string) {
  const [data, setData] = useState<RFMResult[]>([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    supabase
      .from('rfm_results')
      .select('*')
      .eq('business_id', businessId)
      .order('monetary', { ascending: false })
      .then(({ data }) => {
        setData(data ?? [])
        setLoading(false)
      })
  }, [businessId])

  const bySegment = (seg: RFMSegment) => data.filter(d => d.segment_label === seg)
  const champions = bySegment('Champion')
  const atRisk = bySegment('At Risk')
  const cannotLose = bySegment('Cannot Lose')
  const lost = bySegment('Lost')

  return { data, loading, champions, atRisk, cannotLose, lost, bySegment }
}

### SECTION A2 — Stock Velocity Engine + Stockout Prediction

```
Build a real-time stock velocity calculator that predicts days until stockout.

─── POSTGRES FUNCTION ─────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION calculate_stock_velocity(p_business_id UUID)
RETURNS TABLE (
  product_id UUID,
  product_name TEXT,
  current_stock NUMERIC,
  avg_daily_sales NUMERIC,
  days_until_stockout NUMERIC,
  velocity_trend TEXT,
  urgency TEXT
) AS $$
BEGIN
  RETURN QUERY
  WITH daily_sales AS (
    SELECT
      ii.product_id,
      DATE(i.created_at) AS sale_date,
      SUM(ii.quantity) AS qty_sold
    FROM invoice_items ii
    JOIN invoices i ON i.id = ii.invoice_id
    WHERE i.business_id = p_business_id
      AND i.type = 'sale'
      AND i.created_at >= NOW() - INTERVAL '30 days'
    GROUP BY ii.product_id, DATE(i.created_at)
  ),
  velocity AS (
    SELECT
      product_id,
      AVG(qty_sold) AS avg_daily_sales,
      -- Compare last 7 days vs previous 7 days to find trend
      AVG(CASE WHEN sale_date >= NOW() - INTERVAL '7 days' THEN qty_sold END) AS recent_7d,
      AVG(CASE WHEN sale_date BETWEEN NOW() - INTERVAL '14 days'
                              AND NOW() - INTERVAL '7 days' THEN qty_sold END) AS prev_7d
    FROM daily_sales
    GROUP BY product_id
  )
  SELECT
    p.id AS product_id,
    p.name AS product_name,
    p.quantity AS current_stock,
    COALESCE(v.avg_daily_sales, 0) AS avg_daily_sales,
    CASE
      WHEN COALESCE(v.avg_daily_sales, 0) = 0 THEN 9999
      ELSE ROUND(p.quantity / v.avg_daily_sales, 1)
    END AS days_until_stockout,
    CASE
      WHEN v.recent_7d > v.prev_7d * 1.2 THEN 'rising'
      WHEN v.recent_7d < v.prev_7d * 0.8 THEN 'falling'
      ELSE 'stable'
    END AS velocity_trend,
    CASE
      WHEN p.quantity = 0 THEN 'out_of_stock'
      WHEN COALESCE(v.avg_daily_sales, 0) > 0
       AND p.quantity / v.avg_daily_sales <= 3 THEN 'critical'
      WHEN COALESCE(v.avg_daily_sales, 0) > 0
       AND p.quantity / v.avg_daily_sales <= 7 THEN 'warning'
      ELSE 'healthy'
    END AS urgency
  FROM products p
  LEFT JOIN velocity v ON v.product_id = p.id
  WHERE p.business_id = p_business_id
  ORDER BY days_until_stockout ASC;
END;
$$ LANGUAGE plpgsql;

-- Store for fast access
CREATE TABLE IF NOT EXISTS stock_velocity (
  product_id UUID PRIMARY KEY,
  business_id UUID NOT NULL,
  avg_daily_sales NUMERIC DEFAULT 0,
  days_until_stockout NUMERIC DEFAULT 9999,
  velocity_trend TEXT DEFAULT 'stable',
  urgency TEXT DEFAULT 'healthy',
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Refresh every 6 hours
SELECT cron.schedule('velocity-refresh', '0 */6 * * *', $$
  INSERT INTO stock_velocity
  SELECT product_id, p_business_id::UUID, avg_daily_sales,
         days_until_stockout, velocity_trend, urgency, NOW()
  FROM businesses b
  CROSS JOIN LATERAL calculate_stock_velocity(b.id)
  ON CONFLICT (product_id) DO UPDATE SET
    avg_daily_sales = EXCLUDED.avg_daily_sales,
    days_until_stockout = EXCLUDED.days_until_stockout,
    urgency = EXCLUDED.urgency,
    updated_at = NOW()
$$);

### SECTION A3 — Anomaly Detection (Z-Score)

```
Build an anomaly detection system that flags unusual transactions automatically.

─── POSTGRES FUNCTION ─────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION detect_anomalies(p_business_id UUID)
RETURNS TABLE (
  anomaly_type TEXT,
  reference_id UUID,
  reference_type TEXT,
  description TEXT,
  severity TEXT,
  value NUMERIC,
  expected_value NUMERIC,
  z_score NUMERIC,
  detected_at TIMESTAMPTZ
) AS $$
BEGIN
  -- Invoice amount anomalies per customer
  RETURN QUERY
  WITH customer_stats AS (
    SELECT
      contact_id,
      AVG(total_amount) AS avg_amount,
      STDDEV(total_amount) AS std_amount
    FROM invoices
    WHERE business_id = p_business_id AND type = 'sale'
    GROUP BY contact_id
    HAVING COUNT(*) >= 5
  )
  SELECT
    'unusual_invoice_amount'::TEXT,
    i.id,
    'invoice'::TEXT,
    CONCAT('Invoice ₹', i.total_amount::INT, ' for ', c.name,
           ' is unusually ', CASE WHEN i.total_amount > cs.avg_amount THEN 'high' ELSE 'low' END,
           ' (avg: ₹', cs.avg_amount::INT, ')'),
    CASE WHEN ABS((i.total_amount - cs.avg_amount)/NULLIF(cs.std_amount,0)) > 3
         THEN 'high' ELSE 'medium' END,
    i.total_amount,
    cs.avg_amount,
    ROUND(ABS((i.total_amount - cs.avg_amount)/NULLIF(cs.std_amount,0)), 2),
    i.created_at
  FROM invoices i
  JOIN customer_stats cs ON cs.contact_id = i.contact_id
  JOIN contacts c ON c.id = i.contact_id
  WHERE i.business_id = p_business_id
    AND i.created_at >= NOW() - INTERVAL '7 days'
    AND ABS((i.total_amount - cs.avg_amount)/NULLIF(cs.std_amount,0)) > 2

  UNION ALL

  -- Sudden stock drop anomalies
  SELECT
    'sudden_stock_drop'::TEXT,
    sm.id,
    'stock_movement'::TEXT,
    CONCAT(p.name, ' stock dropped by ', ABS(sm.quantity)::INT,
           ' units in one movement'),
    CASE WHEN ABS(sm.quantity) > p.quantity * 0.5 THEN 'high' ELSE 'medium' END,
    ABS(sm.quantity),
    p.quantity * 0.2,
    ROUND(ABS(sm.quantity) / NULLIF(p.quantity * 0.1, 0), 2),
    sm.created_at
  FROM stock_movements sm
  JOIN products p ON p.id = sm.product_id
  WHERE p.business_id = p_business_id
    AND sm.type = 'out'
    AND sm.created_at >= NOW() - INTERVAL '24 hours'
    AND ABS(sm.quantity) > p.quantity * 0.3;
END;
$$ LANGUAGE plpgsql;

-- Anomaly log table
CREATE TABLE IF NOT EXISTS anomaly_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  business_id UUID NOT NULL,
  anomaly_type TEXT,
  reference_id UUID,
  reference_type TEXT,
  description TEXT,
  severity TEXT,
  value NUMERIC,
  expected_value NUMERIC,
  z_score NUMERIC,
  is_dismissed BOOLEAN DEFAULT FALSE,
  detected_at TIMESTAMPTZ DEFAULT NOW()
);

-- Trigger: run anomaly check after every new invoice
CREATE OR REPLACE FUNCTION trg_check_invoice_anomaly()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO anomaly_log (business_id, anomaly_type, reference_id,
    reference_type, description, severity, value, expected_value, z_score)
  SELECT NEW.business_id, anomaly_type, reference_id, reference_type,
    description, severity, value, expected_value, z_score
  FROM detect_anomalies(NEW.business_id)
  WHERE reference_id = NEW.id;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_invoice_anomaly
AFTER INSERT ON invoices
FOR EACH ROW EXECUTE FUNCTION trg_check_invoice_anomaly();

### SECTION A4 — Customer Lifetime Value (CLV)

```
Build a CLV calculator using the simplified BG/NBD model approach.

─── POSTGRES FUNCTION ─────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION calculate_clv(p_business_id UUID)
RETURNS TABLE (
  contact_id UUID,
  contact_name TEXT,
  avg_order_value NUMERIC,
  purchase_frequency NUMERIC,
  customer_lifespan_months NUMERIC,
  clv_estimate NUMERIC,
  clv_tier TEXT
) AS $$
BEGIN
  RETURN QUERY
  WITH customer_metrics AS (
    SELECT
      i.contact_id,
      c.name AS contact_name,
      AVG(i.total_amount) AS avg_order_value,
      -- Purchases per month
      COUNT(i.id)::NUMERIC /
        GREATEST(
          EXTRACT(MONTH FROM AGE(MAX(i.created_at), MIN(i.created_at))) + 1,
          1
        ) AS purchase_frequency,
      -- Months since first purchase
      EXTRACT(MONTH FROM AGE(NOW(), MIN(i.created_at))) + 1 AS customer_lifespan_months
    FROM invoices i
    JOIN contacts c ON c.id = i.contact_id
    WHERE i.business_id = p_business_id
      AND i.type = 'sale'
      AND i.status != 'cancelled'
    GROUP BY i.contact_id, c.name
    HAVING COUNT(i.id) >= 2
  )
  SELECT
    contact_id,
    contact_name,
    ROUND(avg_order_value, 2),
    ROUND(purchase_frequency, 2),
    customer_lifespan_months,
    -- CLV = AOV × frequency per month × projected 12 months × margin (assumed 20%)
    ROUND(avg_order_value * purchase_frequency * 12 * 0.20, 2) AS clv_estimate,
    CASE
      WHEN avg_order_value * purchase_frequency * 12 * 0.20 > 10000 THEN 'Platinum'
      WHEN avg_order_value * purchase_frequency * 12 * 0.20 > 5000  THEN 'Gold'
      WHEN avg_order_value * purchase_frequency * 12 * 0.20 > 1000  THEN 'Silver'
      ELSE 'Bronze'
    END AS clv_tier
  FROM customer_metrics
  ORDER BY clv_estimate DESC;
END;
$$ LANGUAGE plpgsql;

### SECTION A5 — Cash Flow Forecasting Algorithm

```
Build a 30-day cash flow forecast using invoice probability weighting.

─── POSTGRES FUNCTION ─────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION generate_cashflow_forecast(p_business_id UUID)
RETURNS TABLE (
  forecast_date DATE,
  projected_inflow NUMERIC,
  projected_outflow NUMERIC,
  net_cashflow NUMERIC,
  confidence NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  WITH
  -- Calculate each customer's historical payment rate
  customer_pay_rate AS (
    SELECT
      contact_id,
      AVG(CASE WHEN status = 'paid' THEN 1.0 ELSE 0.0 END) AS pay_probability,
      AVG(EXTRACT(DAY FROM updated_at - created_at))
        FILTER (WHERE status = 'paid') AS avg_days_to_pay
    FROM invoices
    WHERE business_id = p_business_id AND type = 'sale'
    GROUP BY contact_id
  ),
  -- Pending invoices weighted by payment probability
  pending_inflows AS (
    SELECT
      -- Estimate payment date = due_date or (created_at + avg_days_to_pay)
      COALESCE(
        i.due_date,
        DATE(i.created_at + INTERVAL '1 day' *
          COALESCE(cpr.avg_days_to_pay, 7))
      ) AS expected_date,
      i.total_amount * COALESCE(cpr.pay_probability, 0.7) AS weighted_amount,
      COALESCE(cpr.pay_probability, 0.7) AS confidence
    FROM invoices i
    LEFT JOIN customer_pay_rate cpr ON cpr.contact_id = i.contact_id
    WHERE i.business_id = p_business_id
      AND i.status = 'pending'
      AND i.type = 'sale'
  ),
  -- Generate 30-day date series
  date_series AS (
    SELECT generate_series(
      CURRENT_DATE,
      CURRENT_DATE + INTERVAL '29 days',
      INTERVAL '1 day'
    )::DATE AS forecast_date
  )
  SELECT
    ds.forecast_date,
    COALESCE(SUM(pi.weighted_amount), 0) AS projected_inflow,
    -- Outflow: purchases + estimated expenses (simplified)
    COALESCE((
      SELECT SUM(i2.total_amount * 0.8)
      FROM invoices i2
      WHERE i2.business_id = p_business_id
        AND i2.type = 'purchase'
        AND i2.status = 'pending'
        AND DATE(i2.due_date) = ds.forecast_date
    ), 0) AS projected_outflow,
    COALESCE(SUM(pi.weighted_amount), 0) -
    COALESCE((
      SELECT SUM(i2.total_amount * 0.8)
      FROM invoices i2
      WHERE i2.business_id = p_business_id
        AND i2.type = 'purchase'
        AND i2.status = 'pending'
        AND DATE(i2.due_date) = ds.forecast_date
    ), 0) AS net_cashflow,
    COALESCE(AVG(pi.confidence), 0.5) AS confidence
  FROM date_series ds
  LEFT JOIN pending_inflows pi ON pi.expected_date = ds.forecast_date
  GROUP BY ds.forecast_date
  ORDER BY ds.forecast_date;
END;
$$ LANGUAGE plpgsql;

─── REACT HOOK ────────────────────────────────────────────────────────────────

// src/hooks/useCashflowForecast.ts
export function useCashflowForecast(businessId: string) {
  const [forecast, setForecast] = useState<ForecastRow[]>([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    supabase.rpc('generate_cashflow_forecast', { p_business_id: businessId })
      .then(({ data }) => {
        setForecast(data ?? [])
        setLoading(false)
      })
  }, [businessId])

  const totalInflow  = forecast.reduce((s, r) => s + r.projected_inflow, 0)
  const totalOutflow = forecast.reduce((s, r) => s + r.projected_outflow, 0)
  const worstDay     = forecast.reduce((a, b) =>
    b.net_cashflow < a.net_cashflow ? b : a, forecast[0])

  return { forecast, loading, totalInflow, totalOutflow, worstDay }
}

3. Gemini AI narrative: feed forecast data to DSS prompt for plain-language summary

export function useDSSStream(analysisType: string, contextData: object) {
  const [text, setText] = useState('')
  const [streaming, setStreaming] = useState(false)

  const run = async () => {
    setStreaming(true)
    setText('')

    const res = await fetch('/functions/v1/dss-analysis', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json',
                 'Authorization': `Bearer ${supabaseKey}` },
      body: JSON.stringify({ analysisType, contextData })
    })

    const reader = res.body!.getReader()
    const decoder = new TextDecoder()

    while (true) {
      const { done, value } = await reader.read()
      if (done) break
      setText(prev => prev + decoder.decode(value))
    }

    setStreaming(false)
  }

  return { text, streaming, run }
}

```
Build a churn prediction model using logistic regression.

CREATE OR REPLACE FUNCTION extract_churn_features(p_business_id UUID)
RETURNS TABLE (

BEGIN
  RETURN QUERY
  SELECT
    c.id,

    AVG(i.total_amount),

    (
      MAX(i.created_at) < NOW() - INTERVAL '60 days' AND

    AND i.business_id = p_business_id AND i.type = 'sale'
  WHERE c.business_id = p_business_id AND c.type = 'customer'
  GROUP BY c.id

END;

from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
import numpy as np
import json

def handler(event, context):
    body = json.loads(event['body'])
    business_id = body['business_id']

    # Fetch features
    features = supabase.rpc('extract_churn_features',
      {'p_business_id': business_id}).execute().data

    df = pd.DataFrame(features)

    feature_cols = ['days_since_last_purchase', 'purchase_count_last_90d',
                    'purchase_count_prev_90d', 'avg_order_value']

    X = df[feature_cols].fillna(0)
    y = df['is_churned'].astype(int)

    if y.sum() < 10:
        return {'statusCode': 400,
                'body': json.dumps({'error': 'Insufficient churn examples'})}

    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    model = LogisticRegression(random_state=42)
    model.fit(X_scaled, y)

    # Predict churn probability for ALL customers
    probs = model.predict_proba(X_scaled)[:, 1]
    df['churn_probability'] = probs
    df['churn_risk'] = pd.cut(probs, bins=[0, 0.3, 0.6, 1.0],
                               labels=['low', 'medium', 'high'])

    # Save predictions
    predictions = df[df['churn_probability'] > 0.4][
      ['contact_id', 'churn_probability', 'churn_risk']
    ].to_dict('records')

    supabase.table('churn_predictions').upsert(predictions,
      on_conflict='contact_id').execute()

    return {'statusCode': 200,
            'body': json.dumps({'at_risk_count': len(predictions)})}

# supabase/functions/product-bundles/index.py
# pip install mlxtend

from mlxtend.frequent_patterns import apriori, association_rules
import pandas as pd

def handler(event, context):
    business_id = json.loads(event['body'])['business_id']

    # Fetch invoice line items
    items = supabase.table('invoice_items') \
        .select('invoice_id, product_id, products(name)') \
        .eq('invoices.business_id', business_id) \
        .execute().data

    df = pd.DataFrame(items)

    # Create basket matrix (invoice × product = True/False)
    basket = df.groupby(['invoice_id', 'product_id'])['product_id'] \
               .count().unstack().fillna(0)
    basket = basket.applymap(lambda x: True if x > 0 else False)

    # Run Apriori
    frequent_items = apriori(basket, min_support=0.02, use_colnames=True)
    rules = association_rules(frequent_items, metric='lift', min_threshold=1.5)
    rules = rules[rules['confidence'] > 0.3].sort_values('lift', ascending=False)

    # Save top 20 bundles
    bundles = []
    for _, row in rules.head(20).iterrows():
        bundles.append({
            'business_id': business_id,
            'antecedent_products': list(row['antecedents']),
            'consequent_products': list(row['consequents']),
            'support': round(row['support'], 3),
            'confidence': round(row['confidence'], 3),
            'lift': round(row['lift'], 3)
        })

    supabase.table('product_bundles').delete() \
        .eq('business_id', business_id).execute()
    supabase.table('product_bundles').insert(bundles).execute()

    return {'statusCode': 200,
            'body': json.dumps({'bundles_found': len(bundles)})}

  // Build context data to send with every VANI request
  const buildContext = useCallback(async () => {
    const [invoices, products, contacts] = await Promise.all([
      supabase.from('invoices')
        .select('id, invoice_number, contact_id, total_amount, status, due_date')
        .eq('business_id', businessId)
        .order('created_at', { ascending: false })
        .limit(10),
      supabase.from('stock_velocity')
        .select('product_id, days_until_stockout, urgency, products(name, quantity)')
        .eq('business_id', businessId)
        .eq('urgency', 'critical')
        .limit(5),
      supabase.from('contacts')
        .select('id, name, type, phone')
        .eq('business_id', businessId)
        .limit(20)
    ])

    return {
      owner_name: ownerName,
      recent_invoices: invoices.data ?? [],
      critical_stock: products.data ?? [],
      contacts: contacts.data ?? [],
      current_time: new Date().toLocaleTimeString('en-IN'),
      current_date: new Date().toLocaleDateString('en-IN')
    }
  }, [businessId, ownerName])

  // Start listening
  const activate = useCallback(() => {
    const SpeechRecognition =
      (window as any).SpeechRecognition ||
      (window as any).webkitSpeechRecognition

    if (!SpeechRecognition) {
      speakVANI("Voice commands are not supported in this browser.")
      return
    }

    const recognition = new SpeechRecognition()
    recognition.continuous = false
    recognition.interimResults = true
    recognition.lang = 'hi-IN'
    recognitionRef.current = recognition

    setState('listening')
    setTranscript('')

    recognition.onresult = (event: any) => {
      const current = Array.from(event.results)
        .map((r: any) => r[0].transcript)
        .join('')
      setTranscript(current)
    }

    recognition.onend = async () => {
      if (!transcript && !recognitionRef.current?.finalTranscript) return
      setState('thinking')
      await processTranscript(transcript)
    }

    recognition.onerror = () => {
      setState('idle')
      speakVANI("I didn't catch that. Please try again.")
    }

    recognition.start()
  }, [transcript])

  // Process transcript through VANI brain
  const processTranscript = async (text: string) => {
    try {
      const context = await buildContext()
      const { data: { session } } = await supabase.auth.getSession()

      const res = await fetch(
        `${import.meta.env.VITE_SUPABASE_URL}/functions/v1/vani-brain`,
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${session?.access_token}`
          },
          body: JSON.stringify({ transcript: text, businessId, contextData: context })
        }
      )

      const response: VANIResponse = await res.json()
      setLastResponse(response)

      // Speak the response
      setState('speaking')
      await speakVANI(response.spoken_response)

      // If proactive note exists, speak it after
      if (response.proactive_note) {
        await new Promise(r => setTimeout(r, 500))
        await speakVANI(response.proactive_note)
      }

      // If needs confirmation, wait
      if (response.requires_confirmation) {
        setPendingAction(response)
        setState('confirming')
        return
      }

      // Execute the action
      await executeVANIAction(response, businessId, updateSettings)
      setState('idle')

    } catch (err) {
      setState('idle')
      speakVANI("Something went wrong. Please try again.")
    }
  }

  // Confirm pending action
  const confirm = async () => {
    if (!pendingAction) return
    await executeVANIAction(pendingAction, businessId, updateSettings)
    setPendingAction(null)
    setState('idle')
    await speakVANI("Done.")
  }

  // Cancel pending action
  const cancel = () => {
    setPendingAction(null)
    setState('idle')
    speakVANI("Cancelled.")
  }

  return {
    state, transcript, lastResponse, pendingAction,
    activate, confirm, cancel
  }
}

// Speak using Web Speech Synthesis
export async function speakVANI(text: string): Promise<void> {
  return new Promise((resolve) => {
    window.speechSynthesis.cancel()
    const utterance = new SpeechSynthesisUtterance(text)
    utterance.lang = 'en-IN'
    utterance.rate = 0.92
    utterance.pitch = 0.95
    utterance.volume = 1
    // Try to use a deeper voice if available
    const voices = window.speechSynthesis.getVoices()
    const preferredVoice = voices.find(v =>
      v.name.includes('Google') && v.lang.includes('en')
    )
    if (preferredVoice) utterance.voice = preferredVoice
    utterance.onend = () => resolve()
    utterance.onerror = () => resolve()
    window.speechSynthesis.speak(utterance)
  })
}

## ─── SECTION V3: VANI ACTION EXECUTOR ──────────────────────────────────────

```
Build the VANI action executor — the layer that translates VANI's JSON response
into actual app actions: navigation, invoice creation, settings changes, etc.

─── src/utils/vaniExecutor.ts ─────────────────────────────────────────────────

import { supabase } from '@/lib/supabase'
import { sendWhatsApp, buildInvoiceMessage } from '@/utils/whatsapp'
import type { VANIResponse } from '@/hooks/useVANI'

export async function executeVANIAction(
  response: VANIResponse,
  businessId: string,
  updateSettings: (key: string, value: any) => void
) {
  const { intent, params } = response

  switch (intent) {

    // ── NAVIGATION ──────────────────────────────────────────────────────────
    case 'navigate_module':
      window.dispatchEvent(new CustomEvent('app:navigate', {
        detail: { module: params.module }
      }))
      break

    // ── CREATE INVOICE ───────────────────────────────────────────────────────
    case 'create_invoice': {
      // Find or create contact
      let contact = null
      if (params.contact_name) {
        const { data } = await supabase
          .from('contacts')
          .select('id, name, phone')
          .eq('business_id', businessId)
          .ilike('name', `%${params.contact_name}%`)
          .single()
        contact = data
      }

      // Resolve product IDs from names
      const resolvedItems = await Promise.all(
        (params.items || []).map(async (item: any) => {
          const { data: product } = await supabase
            .from('products')
            .select('id, name, unit_price')
            .eq('business_id', businessId)
            .ilike('name', `%${item.name}%`)
            .single()

          return {
            product_id: product?.id ?? null,
            product_name: product?.name ?? item.name,
            quantity: item.qty || 1,
            unit_price: item.price || product?.unit_price || 0,
            unit: item.unit || 'pcs'
          }
        })
      )

      // Navigate to invoice module with prefilled data
      window.dispatchEvent(new CustomEvent('app:navigate', {
        detail: {
          module: 'invoices',
          props: {
            mode: 'create',
            prefill: {
              contact_id: contact?.id,
              contact_name: params.contact_name,
              items: resolvedItems,
              discount: params.discount || 0,
              notes: params.notes || ''
            }
          }
        }
      }))
      break
    }

    // ── SEND INVOICE VIA WHATSAPP ────────────────────────────────────────────
    case 'send_invoice_whatsapp': {
      let invoice = null

      if (params.invoice_number) {
        const { data } = await supabase
          .from('invoices')
          .select('*, contacts(name, phone)')
          .eq('business_id', businessId)
          .eq('invoice_number', params.invoice_number)
          .single()
        invoice = data
      } else if (params.contact_name) {
        // Get most recent invoice for this contact
        const { data: contact } = await supabase
          .from('contacts')
          .select('id, name, phone')
          .eq('business_id', businessId)
          .ilike('name', `%${params.contact_name}%`)
          .single()

        if (contact) {
          const { data } = await supabase
            .from('invoices')
            .select('*, contacts(name, phone)')
            .eq('business_id', businessId)
            .eq('contact_id', contact.id)
            .order('created_at', { ascending: false })
            .limit(1)
            .single()
          invoice = data
        }
      }

      if (invoice && invoice.contacts?.phone) {
        const message = buildInvoiceMessage(
          invoice.contacts.name,
          invoice.invoice_number,
          invoice.total_amount,
          invoice.due_date
            ? new Date(invoice.due_date).toLocaleDateString('en-IN')
            : 'as soon as possible'
        )
        sendWhatsApp(invoice.contacts.phone, message)

        // Log the communication
        await supabase.from('communication_log').insert({
          business_id: businessId,
          contact_id: invoice.contact_id,
          channel: 'whatsapp',
          reference_id: invoice.id,
          reference_type: 'invoice',
          message_preview: message.slice(0, 100),
          status: 'sent'
        })
      }
      break
    }

    // ── MARK INVOICE PAID ────────────────────────────────────────────────────
    case 'mark_invoice_paid': {
      let query = supabase
        .from('invoices')
        .update({ status: 'paid', updated_at: new Date().toISOString() })
        .eq('business_id', businessId)

      if (params.invoice_number) {
        query = query.eq('invoice_number', params.invoice_number)
      }
      await query
      break
    }

    // ── CHANGE SETTINGS ──────────────────────────────────────────────────────
    case 'change_setting':
      updateSettings(params.setting_key, params.value)

      // Persist to Supabase
      await supabase
        .from('businesses')
        .update({
          settings: { [params.setting_key]: params.value }
        })
        .eq('id', businessId)
      break

    // ── OPEN SPECIFIC INVOICE ────────────────────────────────────────────────
    case 'open_invoice': {
      const { data: invoice } = await supabase
        .from('invoices')
        .select('id')
        .eq('business_id', businessId)
        .eq('invoice_number', params.invoice_number)
        .single()

      if (invoice) {
        window.dispatchEvent(new CustomEvent('app:navigate', {
          detail: { module: 'invoices', props: { openId: invoice.id } }
        }))
      }
      break
    }

    // ── SET REMINDER ─────────────────────────────────────────────────────────
    case 'set_reminder':
      await supabase.from('reminders').insert({
        business_id: businessId,
        message: params.message,
        remind_at: params.datetime,
        status: 'pending'
      })
      break

    // ── SHOW LOW STOCK ───────────────────────────────────────────────────────
    case 'show_low_stock':
      window.dispatchEvent(new CustomEvent('app:navigate', {
        detail: { module: 'inventory', props: { filter: 'low_stock' } }
      }))
      break

    // ── GET REPORT ───────────────────────────────────────────────────────────
    case 'get_report':
      window.dispatchEvent(new CustomEvent('app:navigate', {
        detail: { module: 'reports', props: { reportType: params.report_type } }
      }))
      break

    default:
      break
  }
}

─── src/hooks/useVANIWakeWord.ts ──────────────────────────────────────────────

export function useVANIWakeWord(onWake: () => void) {
  const recognitionRef = useRef<any>(null)
  const listeningRef = useRef(false)

  const WAKE_WORDS = [
    'hey vani', 'hey vyapari', 'ok vani',
    'aye vani', 'vani', 'hey bani'  // common mishears
  ]

  const startWakeWordDetection = useCallback(() => {
    const SpeechRecognition =
      (window as any).SpeechRecognition ||
      (window as any).webkitSpeechRecognition
    if (!SpeechRecognition || listeningRef.current) return

    const recognition = new SpeechRecognition()
    recognition.continuous = true
    recognition.interimResults = true
    recognition.lang = 'hi-IN'
    recognitionRef.current = recognition
    listeningRef.current = true

    recognition.onresult = (event: any) => {
      const latest = event.results[event.results.length - 1][0].transcript
        .toLowerCase().trim()

      const detected = WAKE_WORDS.some(w => latest.includes(w))
      if (detected) {
        recognition.stop()
        listeningRef.current = false
        onWake()  // Trigger full VANI activation
      }
    }

    recognition.onend = () => {
      // Restart automatically to keep listening
      if (listeningRef.current) {
        setTimeout(() => recognition.start(), 300)
      }
    }

    recognition.start()
  }, [onWake])

  const stopWakeWordDetection = useCallback(() => {
    listeningRef.current = false
    recognitionRef.current?.stop()
  }, [])

  return { startWakeWordDetection, stopWakeWordDetection }
}

```
Run these in Supabase SQL Editor. These power all report data queries.

-- 1. DAILY SALES SUMMARY
CREATE OR REPLACE FUNCTION get_daily_sales_summary(
  p_business_id UUID, p_from DATE, p_to DATE,
  p_group_by TEXT DEFAULT 'day',
  p_include_cancelled BOOLEAN DEFAULT FALSE
) RETURNS TABLE (
  period TEXT, total_revenue NUMERIC, invoice_count INT,
  avg_order_value NUMERIC, total_discount NUMERIC,
  total_tax NUMERIC, net_revenue NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    CASE p_group_by
      WHEN 'day'   THEN TO_CHAR(DATE(i.created_at), 'DD Mon YYYY')
      WHEN 'week'  THEN 'Week ' || TO_CHAR(DATE(i.created_at), 'IW YYYY')
      WHEN 'month' THEN TO_CHAR(DATE(i.created_at), 'Mon YYYY')
    END AS period,
    SUM(i.total_amount), COUNT(i.id)::INT,
    AVG(i.total_amount), COALESCE(SUM(i.discount_amount),0),
    SUM(i.tax_amount), SUM(i.total_amount - i.tax_amount)
  FROM invoices i
  WHERE i.business_id = p_business_id AND i.type = 'sale'
    AND DATE(i.created_at) BETWEEN p_from AND p_to
    AND (p_include_cancelled OR i.status != 'cancelled')
  GROUP BY 1 ORDER BY MIN(i.created_at);
END; $$ LANGUAGE plpgsql;

-- 2. HOURLY SALES ANALYSIS
CREATE OR REPLACE FUNCTION get_hourly_sales(
  p_business_id UUID, p_date DATE
) RETURNS TABLE (hour_label TEXT, revenue NUMERIC, invoice_count INT) AS $$
BEGIN
  RETURN QUERY
  SELECT
    TO_CHAR(gs.hour_start AT TIME ZONE 'Asia/Kolkata', 'HH12 AM'),
    COALESCE(SUM(i.total_amount), 0),
    COUNT(i.id)::INT
  FROM generate_series(
    p_date::TIMESTAMPTZ, (p_date+1)::TIMESTAMPTZ - INTERVAL '1 hour',
    INTERVAL '1 hour'
  ) gs(hour_start)
  LEFT JOIN invoices i
    ON DATE_TRUNC('hour', i.created_at AT TIME ZONE 'Asia/Kolkata') =
       gs.hour_start AT TIME ZONE 'Asia/Kolkata'
    AND i.business_id = p_business_id AND i.type = 'sale'
  GROUP BY gs.hour_start ORDER BY gs.hour_start;
END; $$ LANGUAGE plpgsql;

-- 3. PAYMENT MODE ANALYSIS
CREATE OR REPLACE FUNCTION get_payment_mode_analysis(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  payment_mode TEXT, total_amount NUMERIC,
  invoice_count INT, percentage NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  WITH totals AS (
    SELECT SUM(total_amount) AS grand_total FROM invoices
    WHERE business_id=p_business_id AND type='sale'
      AND status!='cancelled' AND DATE(created_at) BETWEEN p_from AND p_to
  )
  SELECT COALESCE(i.payment_mode,'Cash'),
    SUM(i.total_amount), COUNT(i.id)::INT,
    ROUND(SUM(i.total_amount)/NULLIF(t.grand_total,0)*100,1)
  FROM invoices i, totals t
  WHERE i.business_id=p_business_id AND i.type='sale'
    AND i.status!='cancelled' AND DATE(i.created_at) BETWEEN p_from AND p_to
  GROUP BY i.payment_mode, t.grand_total
  ORDER BY SUM(i.total_amount) DESC;
END; $$ LANGUAGE plpgsql;

-- 4. TAX / GST SUMMARY
CREATE OR REPLACE FUNCTION get_gst_summary(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  tax_rate NUMERIC, taxable_amount NUMERIC, tax_collected NUMERIC,
  invoice_count INT, b2b_amount NUMERIC, b2c_amount NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT ii.tax_rate,
    SUM(ii.quantity*ii.unit_price),
    SUM(ii.quantity*ii.unit_price*ii.tax_rate/100),
    COUNT(DISTINCT i.id)::INT,
    SUM(CASE WHEN c.gstin IS NOT NULL THEN ii.quantity*ii.unit_price ELSE 0 END),
    SUM(CASE WHEN c.gstin IS NULL     THEN ii.quantity*ii.unit_price ELSE 0 END)
  FROM invoice_items ii
  JOIN invoices i ON i.id=ii.invoice_id
  JOIN contacts c ON c.id=i.contact_id
  WHERE i.business_id=p_business_id AND i.type='sale'
    AND i.status!='cancelled'
    AND DATE(i.created_at) BETWEEN p_from AND p_to
  GROUP BY ii.tax_rate ORDER BY ii.tax_rate;
END; $$ LANGUAGE plpgsql;

-- 5. LOW STOCK ALERT
CREATE OR REPLACE FUNCTION get_low_stock_report(
  p_business_id UUID,
  p_urgency TEXT[] DEFAULT ARRAY['critical','warning','out_of_stock'],
  p_below_reorder_only BOOLEAN DEFAULT FALSE
) RETURNS TABLE (
  product_name TEXT, sku TEXT, category TEXT, current_stock NUMERIC,
  reorder_level NUMERIC, days_until_stockout NUMERIC,
  urgency TEXT, supplier_name TEXT, stock_value NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT p.name, p.sku, p.category, p.quantity, p.reorder_level,
    sv.days_until_stockout, sv.urgency,
    c.name AS supplier_name,
    p.quantity * p.cost_price AS stock_value
  FROM products p
  LEFT JOIN stock_velocity sv ON sv.product_id=p.id
  LEFT JOIN contacts c ON c.id=p.supplier_id
  WHERE p.business_id=p_business_id
    AND sv.urgency=ANY(p_urgency)
    AND (NOT p_below_reorder_only OR p.quantity<=p.reorder_level)
  ORDER BY sv.days_until_stockout ASC NULLS LAST;
END; $$ LANGUAGE plpgsql;

-- 6. DEAD STOCK ANALYSIS
CREATE OR REPLACE FUNCTION get_dead_stock(
  p_business_id UUID,
  p_no_movement_days INT DEFAULT 60,
  p_min_stock_value NUMERIC DEFAULT 0
) RETURNS TABLE (
  product_name TEXT, category TEXT, current_stock NUMERIC,
  stock_value NUMERIC, last_sale_date DATE, days_without_sale INT,
  supplier_name TEXT
) AS $$
BEGIN
  RETURN QUERY
  SELECT p.name, p.category, p.quantity,
    p.quantity*p.cost_price AS stock_value,
    MAX(DATE(i.created_at)) FILTER (WHERE i.type='sale') AS last_sale_date,
    EXTRACT(DAY FROM NOW()-MAX(i.created_at) FILTER (WHERE i.type='sale'))::INT,
    c.name AS supplier_name
  FROM products p
  LEFT JOIN invoice_items ii ON ii.product_id=p.id
  LEFT JOIN invoices i ON i.id=ii.invoice_id
  LEFT JOIN contacts c ON c.id=p.supplier_id
  WHERE p.business_id=p_business_id AND p.quantity>0
  GROUP BY p.id, p.name, p.category, p.quantity, p.cost_price, c.name
  HAVING COALESCE(EXTRACT(DAY FROM NOW()-MAX(i.created_at)
    FILTER (WHERE i.type='sale')),9999) >= p_no_movement_days
    AND p.quantity*p.cost_price >= p_min_stock_value
  ORDER BY days_without_sale DESC;
END; $$ LANGUAGE plpgsql;

-- 7. INVENTORY VALUATION
CREATE OR REPLACE FUNCTION get_inventory_valuation(p_business_id UUID)
RETURNS TABLE (
  category TEXT, product_count INT, total_units NUMERIC,
  total_value NUMERIC, percentage_of_total NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  WITH totals AS (
    SELECT SUM(quantity*cost_price) AS grand
    FROM products WHERE business_id=p_business_id AND quantity>0
  )
  SELECT COALESCE(p.category,'Uncategorised'),
    COUNT(p.id)::INT, SUM(p.quantity),
    SUM(p.quantity*p.cost_price),
    ROUND(SUM(p.quantity*p.cost_price)/NULLIF(t.grand,0)*100,1)
  FROM products p, totals t
  WHERE p.business_id=p_business_id AND p.quantity>0
  GROUP BY p.category, t.grand
  ORDER BY SUM(p.quantity*p.cost_price) DESC;
END; $$ LANGUAGE plpgsql;

-- 8. P&L STATEMENT
CREATE OR REPLACE FUNCTION get_pl_statement(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (line_item TEXT, amount NUMERIC, category TEXT, sort_order INT) AS $$
DECLARE
  v_gross NUMERIC; v_disc NUMERIC; v_cogs NUMERIC;
BEGIN
  SELECT COALESCE(SUM(total_amount),0) INTO v_gross FROM invoices
  WHERE business_id=p_business_id AND type='sale'
    AND DATE(created_at) BETWEEN p_from AND p_to AND status!='cancelled';
  SELECT COALESCE(SUM(discount_amount),0) INTO v_disc FROM invoices
  WHERE business_id=p_business_id AND type='sale'
    AND DATE(created_at) BETWEEN p_from AND p_to AND status!='cancelled';
  SELECT COALESCE(SUM(total_amount),0) INTO v_cogs FROM invoices
  WHERE business_id=p_business_id AND type='purchase'
    AND DATE(created_at) BETWEEN p_from AND p_to AND status!='cancelled';
  RETURN QUERY VALUES
    ('Gross Sales',            v_gross,            'revenue',   1),
    ('Less: Discounts',        -v_disc,            'deduction', 2),
    ('Net Sales',              v_gross-v_disc,     'subtotal',  3),
    ('Cost of Goods Sold',     -v_cogs,            'expense',   4),
    ('Gross Profit',           v_gross-v_disc-v_cogs, 'subtotal', 5),
    ('Net Profit / (Loss)',    v_gross-v_disc-v_cogs, 'total',  6);
END; $$ LANGUAGE plpgsql;

-- 9. SUPPLIER PERFORMANCE
CREATE OR REPLACE FUNCTION get_supplier_performance(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  supplier_name TEXT, total_orders INT, total_value NUMERIC,
  avg_order_value NUMERIC, last_order_date DATE
) AS $$
BEGIN
  RETURN QUERY
  SELECT c.name, COUNT(i.id)::INT, SUM(i.total_amount),
    AVG(i.total_amount), MAX(DATE(i.created_at))
  FROM invoices i JOIN contacts c ON c.id=i.contact_id
  WHERE i.business_id=p_business_id AND i.type='purchase'
    AND DATE(i.created_at) BETWEEN p_from AND p_to
  GROUP BY c.id, c.name ORDER BY SUM(i.total_amount) DESC;
END; $$ LANGUAGE plpgsql;

-- 10. SALES BY EMPLOYEE
CREATE OR REPLACE FUNCTION get_sales_by_employee(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  employee_name TEXT, invoice_count INT,
  total_revenue NUMERIC, avg_order_value NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT pr.full_name, COUNT(i.id)::INT,
    SUM(i.total_amount), AVG(i.total_amount)
  FROM invoices i JOIN profiles pr ON pr.id=i.created_by
  WHERE i.business_id=p_business_id
    AND DATE(i.created_at) BETWEEN p_from AND p_to
    AND i.type='sale' AND i.status!='cancelled'
  GROUP BY pr.full_name, i.created_by
  ORDER BY SUM(i.total_amount) DESC;
END; $$ LANGUAGE plpgsql;

```
Run these in Supabase SQL Editor. These power all report data queries.

-- 1. DAILY SALES SUMMARY
CREATE OR REPLACE FUNCTION get_daily_sales_summary(
  p_business_id UUID, p_from DATE, p_to DATE,
  p_group_by TEXT DEFAULT 'day',
  p_include_cancelled BOOLEAN DEFAULT FALSE
) RETURNS TABLE (
  period TEXT, total_revenue NUMERIC, invoice_count INT,
  avg_order_value NUMERIC, total_discount NUMERIC,
  total_tax NUMERIC, net_revenue NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    CASE p_group_by
      WHEN 'day'   THEN TO_CHAR(DATE(i.created_at), 'DD Mon YYYY')
      WHEN 'week'  THEN 'Week ' || TO_CHAR(DATE(i.created_at), 'IW YYYY')
      WHEN 'month' THEN TO_CHAR(DATE(i.created_at), 'Mon YYYY')
    END AS period,
    SUM(i.total_amount), COUNT(i.id)::INT,
    AVG(i.total_amount), COALESCE(SUM(i.discount_amount),0),
    SUM(i.tax_amount), SUM(i.total_amount - i.tax_amount)
  FROM invoices i
  WHERE i.business_id = p_business_id AND i.type = 'sale'
    AND DATE(i.created_at) BETWEEN p_from AND p_to
    AND (p_include_cancelled OR i.status != 'cancelled')
  GROUP BY 1 ORDER BY MIN(i.created_at);
END; $$ LANGUAGE plpgsql;

-- 2. HOURLY SALES ANALYSIS
CREATE OR REPLACE FUNCTION get_hourly_sales(
  p_business_id UUID, p_date DATE
) RETURNS TABLE (hour_label TEXT, revenue NUMERIC, invoice_count INT) AS $$
BEGIN
  RETURN QUERY
  SELECT
    TO_CHAR(gs.hour_start AT TIME ZONE 'Asia/Kolkata', 'HH12 AM'),
    COALESCE(SUM(i.total_amount), 0),
    COUNT(i.id)::INT
  FROM generate_series(
    p_date::TIMESTAMPTZ, (p_date+1)::TIMESTAMPTZ - INTERVAL '1 hour',
    INTERVAL '1 hour'
  ) gs(hour_start)
  LEFT JOIN invoices i
    ON DATE_TRUNC('hour', i.created_at AT TIME ZONE 'Asia/Kolkata') =
       gs.hour_start AT TIME ZONE 'Asia/Kolkata'
    AND i.business_id = p_business_id AND i.type = 'sale'
  GROUP BY gs.hour_start ORDER BY gs.hour_start;
END; $$ LANGUAGE plpgsql;

-- 3. PAYMENT MODE ANALYSIS
CREATE OR REPLACE FUNCTION get_payment_mode_analysis(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  payment_mode TEXT, total_amount NUMERIC,
  invoice_count INT, percentage NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  WITH totals AS (
    SELECT SUM(total_amount) AS grand_total FROM invoices
    WHERE business_id=p_business_id AND type='sale'
      AND status!='cancelled' AND DATE(created_at) BETWEEN p_from AND p_to
  )
  SELECT COALESCE(i.payment_mode,'Cash'),
    SUM(i.total_amount), COUNT(i.id)::INT,
    ROUND(SUM(i.total_amount)/NULLIF(t.grand_total,0)*100,1)
  FROM invoices i, totals t
  WHERE i.business_id=p_business_id AND i.type='sale'
    AND i.status!='cancelled' AND DATE(i.created_at) BETWEEN p_from AND p_to
  GROUP BY i.payment_mode, t.grand_total
  ORDER BY SUM(i.total_amount) DESC;
END; $$ LANGUAGE plpgsql;

-- 4. TAX / GST SUMMARY
CREATE OR REPLACE FUNCTION get_gst_summary(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  tax_rate NUMERIC, taxable_amount NUMERIC, tax_collected NUMERIC,
  invoice_count INT, b2b_amount NUMERIC, b2c_amount NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT ii.tax_rate,
    SUM(ii.quantity*ii.unit_price),
    SUM(ii.quantity*ii.unit_price*ii.tax_rate/100),
    COUNT(DISTINCT i.id)::INT,
    SUM(CASE WHEN c.gstin IS NOT NULL THEN ii.quantity*ii.unit_price ELSE 0 END),
    SUM(CASE WHEN c.gstin IS NULL     THEN ii.quantity*ii.unit_price ELSE 0 END)
  FROM invoice_items ii
  JOIN invoices i ON i.id=ii.invoice_id
  JOIN contacts c ON c.id=i.contact_id
  WHERE i.business_id=p_business_id AND i.type='sale'
    AND i.status!='cancelled'
    AND DATE(i.created_at) BETWEEN p_from AND p_to
  GROUP BY ii.tax_rate ORDER BY ii.tax_rate;
END; $$ LANGUAGE plpgsql;

-- 5. LOW STOCK ALERT
CREATE OR REPLACE FUNCTION get_low_stock_report(
  p_business_id UUID,
  p_urgency TEXT[] DEFAULT ARRAY['critical','warning','out_of_stock'],
  p_below_reorder_only BOOLEAN DEFAULT FALSE
) RETURNS TABLE (
  product_name TEXT, sku TEXT, category TEXT, current_stock NUMERIC,
  reorder_level NUMERIC, days_until_stockout NUMERIC,
  urgency TEXT, supplier_name TEXT, stock_value NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT p.name, p.sku, p.category, p.quantity, p.reorder_level,
    sv.days_until_stockout, sv.urgency,
    c.name AS supplier_name,
    p.quantity * p.cost_price AS stock_value
  FROM products p
  LEFT JOIN stock_velocity sv ON sv.product_id=p.id
  LEFT JOIN contacts c ON c.id=p.supplier_id
  WHERE p.business_id=p_business_id
    AND sv.urgency=ANY(p_urgency)
    AND (NOT p_below_reorder_only OR p.quantity<=p.reorder_level)
  ORDER BY sv.days_until_stockout ASC NULLS LAST;
END; $$ LANGUAGE plpgsql;

-- 6. DEAD STOCK ANALYSIS
CREATE OR REPLACE FUNCTION get_dead_stock(
  p_business_id UUID,
  p_no_movement_days INT DEFAULT 60,
  p_min_stock_value NUMERIC DEFAULT 0
) RETURNS TABLE (
  product_name TEXT, category TEXT, current_stock NUMERIC,
  stock_value NUMERIC, last_sale_date DATE, days_without_sale INT,
  supplier_name TEXT
) AS $$
BEGIN
  RETURN QUERY
  SELECT p.name, p.category, p.quantity,
    p.quantity*p.cost_price AS stock_value,
    MAX(DATE(i.created_at)) FILTER (WHERE i.type='sale') AS last_sale_date,
    EXTRACT(DAY FROM NOW()-MAX(i.created_at) FILTER (WHERE i.type='sale'))::INT,
    c.name AS supplier_name
  FROM products p
  LEFT JOIN invoice_items ii ON ii.product_id=p.id
  LEFT JOIN invoices i ON i.id=ii.invoice_id
  LEFT JOIN contacts c ON c.id=p.supplier_id
  WHERE p.business_id=p_business_id AND p.quantity>0
  GROUP BY p.id, p.name, p.category, p.quantity, p.cost_price, c.name
  HAVING COALESCE(EXTRACT(DAY FROM NOW()-MAX(i.created_at)
    FILTER (WHERE i.type='sale')),9999) >= p_no_movement_days
    AND p.quantity*p.cost_price >= p_min_stock_value
  ORDER BY days_without_sale DESC;
END; $$ LANGUAGE plpgsql;

-- 7. INVENTORY VALUATION
CREATE OR REPLACE FUNCTION get_inventory_valuation(p_business_id UUID)
RETURNS TABLE (
  category TEXT, product_count INT, total_units NUMERIC,
  total_value NUMERIC, percentage_of_total NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  WITH totals AS (
    SELECT SUM(quantity*cost_price) AS grand
    FROM products WHERE business_id=p_business_id AND quantity>0
  )
  SELECT COALESCE(p.category,'Uncategorised'),
    COUNT(p.id)::INT, SUM(p.quantity),
    SUM(p.quantity*p.cost_price),
    ROUND(SUM(p.quantity*p.cost_price)/NULLIF(t.grand,0)*100,1)
  FROM products p, totals t
  WHERE p.business_id=p_business_id AND p.quantity>0
  GROUP BY p.category, t.grand
  ORDER BY SUM(p.quantity*p.cost_price) DESC;
END; $$ LANGUAGE plpgsql;

-- 8. P&L STATEMENT
CREATE OR REPLACE FUNCTION get_pl_statement(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (line_item TEXT, amount NUMERIC, category TEXT, sort_order INT) AS $$
DECLARE
  v_gross NUMERIC; v_disc NUMERIC; v_cogs NUMERIC;
BEGIN
  SELECT COALESCE(SUM(total_amount),0) INTO v_gross FROM invoices
  WHERE business_id=p_business_id AND type='sale'
    AND DATE(created_at) BETWEEN p_from AND p_to AND status!='cancelled';
  SELECT COALESCE(SUM(discount_amount),0) INTO v_disc FROM invoices
  WHERE business_id=p_business_id AND type='sale'
    AND DATE(created_at) BETWEEN p_from AND p_to AND status!='cancelled';
  SELECT COALESCE(SUM(total_amount),0) INTO v_cogs FROM invoices
  WHERE business_id=p_business_id AND type='purchase'
    AND DATE(created_at) BETWEEN p_from AND p_to AND status!='cancelled';
  RETURN QUERY VALUES
    ('Gross Sales',            v_gross,            'revenue',   1),
    ('Less: Discounts',        -v_disc,            'deduction', 2),
    ('Net Sales',              v_gross-v_disc,     'subtotal',  3),
    ('Cost of Goods Sold',     -v_cogs,            'expense',   4),
    ('Gross Profit',           v_gross-v_disc-v_cogs, 'subtotal', 5),
    ('Net Profit / (Loss)',    v_gross-v_disc-v_cogs, 'total',  6);
END; $$ LANGUAGE plpgsql;

-- 9. SUPPLIER PERFORMANCE
CREATE OR REPLACE FUNCTION get_supplier_performance(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  supplier_name TEXT, total_orders INT, total_value NUMERIC,
  avg_order_value NUMERIC, last_order_date DATE
) AS $$
BEGIN
  RETURN QUERY
  SELECT c.name, COUNT(i.id)::INT, SUM(i.total_amount),
    AVG(i.total_amount), MAX(DATE(i.created_at))
  FROM invoices i JOIN contacts c ON c.id=i.contact_id
  WHERE i.business_id=p_business_id AND i.type='purchase'
    AND DATE(i.created_at) BETWEEN p_from AND p_to
  GROUP BY c.id, c.name ORDER BY SUM(i.total_amount) DESC;
END; $$ LANGUAGE plpgsql;

-- 10. SALES BY EMPLOYEE
CREATE OR REPLACE FUNCTION get_sales_by_employee(
  p_business_id UUID, p_from DATE, p_to DATE
) RETURNS TABLE (
  employee_name TEXT, invoice_count INT,
  total_revenue NUMERIC, avg_order_value NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  SELECT pr.full_name, COUNT(i.id)::INT,
    SUM(i.total_amount), AVG(i.total_amount)
  FROM invoices i JOIN profiles pr ON pr.id=i.created_by
  WHERE i.business_id=p_business_id
    AND DATE(i.created_at) BETWEEN p_from AND p_to
    AND i.type='sale' AND i.status!='cancelled'
  GROUP BY pr.full_name, i.created_by
  ORDER BY SUM(i.total_amount) DESC;
END; $$ LANGUAGE plpgsql;

// src/hooks/useSmartAutocomplete.ts
import { useState, useEffect, useRef } from 'react'
import { supabase } from '@/lib/supabase'

export function useSmartAutocomplete(businessId: string) {
  const [query, setQuery] = useState('')
  const [results, setResults] = useState<AutocompleteItem[]>([])
  const [loading, setLoading] = useState(false)
  const debounceRef = useRef<NodeJS.Timeout>()

  useEffect(() => {
    if (query.length < 1) { setResults([]); return }

    clearTimeout(debounceRef.current)
    debounceRef.current = setTimeout(async () => {
      setLoading(true)

      // Rank by: frequency billed + fuzzy name match + stock availability
      const { data } = await supabase.rpc('search_products_smart', {
        p_business_id: businessId,
        p_query: query,
        p_limit: 8
      })

      setResults(data ?? [])
      setLoading(false)
    }, 150) // 150ms debounce — fast enough for typing
  }, [query, businessId])

  return { query, setQuery, results, loading }
}

// POSTGRES FUNCTION — run in Supabase SQL Editor:
CREATE OR REPLACE FUNCTION search_products_smart(
  p_business_id UUID,
  p_query TEXT,
  p_limit INT DEFAULT 8
) RETURNS TABLE (
  product_id UUID, name TEXT, unit_price NUMERIC,
  cost_price NUMERIC, quantity NUMERIC, unit TEXT,
  tax_rate NUMERIC, bill_frequency INT,
  stock_status TEXT
) AS $$
BEGIN
  RETURN QUERY
  WITH frequency AS (
    SELECT ii.product_id, COUNT(*)::INT AS freq
    FROM invoice_items ii
    JOIN invoices i ON i.id = ii.invoice_id
    WHERE i.business_id = p_business_id
      AND i.created_at >= NOW() - INTERVAL '90 days'
    GROUP BY ii.product_id
  )
  SELECT
    p.id, p.name, p.unit_price, p.cost_price,
    p.quantity, p.unit, p.tax_rate,
    COALESCE(f.freq, 0) AS bill_frequency,
    CASE
      WHEN p.quantity = 0 THEN 'out_of_stock'
      WHEN p.quantity <= p.reorder_level THEN 'low_stock'
      ELSE 'in_stock'
    END AS stock_status
  FROM products p
  LEFT JOIN frequency f ON f.product_id = p.id
  WHERE p.business_id = p_business_id
    AND p.name ILIKE '%' || p_query || '%'
  ORDER BY
    CASE WHEN LOWER(p.name) LIKE LOWER(p_query) || '%' THEN 0 ELSE 1 END,
    COALESCE(f.freq, 0) DESC,
    p.name ASC
  LIMIT p_limit;
END; $$ LANGUAGE plpgsql;

POSTGRES FUNCTION:
CREATE OR REPLACE FUNCTION check_duplicate_invoice(
  p_business_id UUID,
  p_contact_id UUID,
  p_total_amount NUMERIC,
  p_item_count INT
) RETURNS TABLE (
  invoice_id UUID, invoice_number TEXT,
  created_at TIMESTAMPTZ, total_amount NUMERIC,
  similarity_score INT
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    i.id, i.invoice_number, i.created_at, i.total_amount,
    (
      CASE WHEN ABS(i.total_amount - p_total_amount) < p_total_amount * 0.05
           THEN 50 ELSE 0 END +
      CASE WHEN (SELECT COUNT(*) FROM invoice_items WHERE invoice_id = i.id)
                = p_item_count THEN 30 ELSE 0 END +
      CASE WHEN i.created_at > NOW() - INTERVAL '2 hours' THEN 20 ELSE 0 END
    )::INT AS similarity_score
  FROM invoices i
  WHERE i.business_id = p_business_id
    AND i.contact_id = p_contact_id
    AND i.created_at > NOW() - INTERVAL '24 hours'
    AND i.status != 'cancelled'
  HAVING (
    CASE WHEN ABS(i.total_amount - p_total_amount) < p_total_amount * 0.05
         THEN 50 ELSE 0 END +
    CASE WHEN (SELECT COUNT(*) FROM invoice_items WHERE invoice_id = i.id)
              = p_item_count THEN 30 ELSE 0 END +
    CASE WHEN i.created_at > NOW() - INTERVAL '2 hours' THEN 20 ELSE 0 END
  ) >= 50
  ORDER BY i.created_at DESC;
END; $$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION get_customer_credit_status(
  p_business_id UUID, p_contact_id UUID
) RETURNS TABLE (
  contact_name TEXT, outstanding_amount NUMERIC,
  credit_limit NUMERIC, available_credit NUMERIC,
  credit_used_pct NUMERIC, last_invoice_number TEXT,
  last_invoice_amount NUMERIC, last_invoice_days_ago INT,
  avg_payment_days NUMERIC, payment_label TEXT
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    c.name,
    COALESCE(SUM(i.total_amount) FILTER (WHERE i.status = 'pending'), 0) AS outstanding,
    COALESCE(c.credit_limit, 0),
    GREATEST(0, COALESCE(c.credit_limit, 0) -
      COALESCE(SUM(i.total_amount) FILTER (WHERE i.status = 'pending'), 0)),
    CASE WHEN COALESCE(c.credit_limit, 0) > 0
         THEN ROUND(COALESCE(SUM(i.total_amount) FILTER
           (WHERE i.status = 'pending'), 0) / c.credit_limit * 100, 0)
         ELSE 0 END,
    (SELECT invoice_number FROM invoices
     WHERE contact_id = p_contact_id AND business_id = p_business_id
     ORDER BY created_at DESC LIMIT 1),
    (SELECT total_amount FROM invoices
     WHERE contact_id = p_contact_id AND business_id = p_business_id
     ORDER BY created_at DESC LIMIT 1),
    EXTRACT(DAY FROM NOW() - (
      SELECT created_at FROM invoices
      WHERE contact_id = p_contact_id AND business_id = p_business_id
      ORDER BY created_at DESC LIMIT 1))::INT,
    ROUND(AVG(EXTRACT(DAY FROM updated_at - created_at))
      FILTER (WHERE i.status = 'paid'), 1),
    CASE
      WHEN AVG(EXTRACT(DAY FROM updated_at - created_at))
           FILTER (WHERE i.status = 'paid') <= 7 THEN 'Excellent payer'
      WHEN AVG(EXTRACT(DAY FROM updated_at - created_at))
           FILTER (WHERE i.status = 'paid') <= 15 THEN 'Good payer'
      WHEN AVG(EXTRACT(DAY FROM updated_at - created_at))
           FILTER (WHERE i.status = 'paid') <= 30 THEN 'Average payer'
      ELSE 'Slow payer'
    END
  FROM contacts c
  LEFT JOIN invoices i ON i.contact_id = c.id AND i.business_id = p_business_id
  WHERE c.id = p_contact_id AND c.business_id = p_business_id
  GROUP BY c.id, c.name, c.credit_limit;
END; $$ LANGUAGE plpgsql;

// supabase/functions/recurring-invoices/index.ts
Deno.serve(async () => {
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )

  // Find all schedules due today or overdue
  const { data: due } = await supabase
    .from('recurring_schedule')
    .select('*, invoices(*)')
    .eq('is_active', true)
    .lte('next_run_date', new Date().toISOString().split('T')[0])

  for (const schedule of due ?? []) {
    const template = schedule.invoices

    // Clone the invoice
    const { data: newInvoice } = await supabase
      .from('invoices')
      .insert({
        business_id: template.business_id,
        contact_id: template.contact_id,
        type: template.type,
        status: 'pending',
        notes: template.notes,
        payment_mode: template.payment_mode,
        discount_amount: template.discount_amount,
        recurrence_parent_id: template.id,
        created_via: 'recurring'
      })
      .select()
      .single()

    // Clone line items
    const { data: items } = await supabase
      .from('invoice_items')
      .select('*')
      .eq('invoice_id', template.id)

    await supabase.from('invoice_items').insert(
      items!.map(item => ({
        invoice_id: newInvoice!.id,
        product_id: item.product_id,
        quantity: item.quantity,
        unit_price: item.unit_price,
        tax_rate: item.tax_rate
      }))
    )

    // Calculate next run date
    const nextDate = new Date(schedule.next_run_date)
    if (schedule.interval === 'weekly') nextDate.setDate(nextDate.getDate() + 7)
    else if (schedule.interval === 'biweekly') nextDate.setDate(nextDate.getDate() + 14)
    else if (schedule.interval === 'monthly') nextDate.setMonth(nextDate.getMonth() + 1)
    else if (schedule.interval === 'quarterly') nextDate.setMonth(nextDate.getMonth() + 3)

    await supabase.from('recurring_schedule')
      .update({ next_run_date: nextDate.toISOString().split('T')[0],
                last_run_date: new Date().toISOString().split('T')[0] })
      .eq('id', schedule.id)
  }

  return new Response(JSON.stringify({ processed: due?.length ?? 0 }))
})

PURPOSE: Automatically fill the correct GST components based on:

  - Whether it's intra-state or inter-state (CGST+SGST vs IGST)

// src/utils/gstCalculator.ts
interface GSTBreakdown {
  taxable_amount: number
  cgst_rate: number
  cgst_amount: number
  sgst_rate: number
  sgst_amount: number
  igst_rate: number
  igst_amount: number
  total_tax: number
  is_interstate: boolean
}

export function calculateGST(
  amount: number,
  taxRate: number,
  businessState: string,
  customerGstin?: string,
  customerState?: string
): GSTBreakdown {
  const taxable = amount
  const isB2B = Boolean(customerGstin)

  // Determine inter-state from first 2 digits of GSTIN (state code)
  const businessStateCode = getStateCodeFromGSTIN(/* business GSTIN */)
  const customerStateCode = customerGstin?.slice(0, 2) ?? businessStateCode
  const isInterstate = isB2B && businessStateCode !== customerStateCode

  if (isInterstate) {
    const igst = (taxable * taxRate) / 100
    return {
      taxable_amount: taxable,
      cgst_rate: 0, cgst_amount: 0,
      sgst_rate: 0, sgst_amount: 0,
      igst_rate: taxRate, igst_amount: igst,
      total_tax: igst, is_interstate: true
    }
  } else {
    const halfRate = taxRate / 2
    const cgst = (taxable * halfRate) / 100
    const sgst = (taxable * halfRate) / 100
    return {
      taxable_amount: taxable,
      cgst_rate: halfRate, cgst_amount: cgst,
      sgst_rate: halfRate, sgst_amount: sgst,
      igst_rate: 0, igst_amount: 0,
      total_tax: cgst + sgst, is_interstate: false
    }
  }
}

function getStateCodeFromGSTIN(gstin: string): string {
  return gstin?.slice(0, 2) ?? '27' // default Maharashtra
}

export function usePaymentScore(customerId: string, invoiceAmount: number) {
  const [score, setScore] = useState<PaymentScore | null>(null)
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    if (!customerId || invoiceAmount < 1) return

    const cacheKey = `payment_score_${customerId}_${Math.round(invoiceAmount/100)*100}`

    const fetch = async () => {
      setLoading(true)

      // Check cache first
      const { data: cached } = await supabase.from('ai_cache')
        .select('response').eq('cache_key', cacheKey)
        .gt('expires_at', new Date().toISOString()).single()

      if (cached) { setScore(cached.response); setLoading(false); return }

      // Fetch customer data
      const { data: customer } = await supabase
        .rpc('get_customer_credit_status', {
          p_business_id: businessId, p_contact_id: customerId
        })

      // Call Gemini
      const res = await supabase.functions.invoke('payment-score', {
        body: { customerData: customer?.[0], invoiceAmount }
      })
      const result = res.data

      // Cache for 30 minutes
      await supabase.from('ai_cache').upsert({
        cache_key: cacheKey, response: result,
        expires_at: new Date(Date.now() + 30 * 60 * 1000).toISOString()
      })

      setScore(result)
      setLoading(false)
    }

    const debounce = setTimeout(fetch, 800)
    return () => clearTimeout(debounce)
  }, [customerId, invoiceAmount])

  return { score, loading }

CREATE OR REPLACE FUNCTION get_upsell_suggestions(
  p_business_id UUID,
  p_contact_id UUID,
  p_current_product_ids UUID[],
  p_limit INT DEFAULT 3
) RETURNS TABLE (
  product_id UUID, product_name TEXT, unit_price NUMERIC,
  reason TEXT, co_purchase_rate NUMERIC
) AS $$
BEGIN
  RETURN QUERY
  WITH co_purchased AS (
    -- Products bought in same invoices as current items
    SELECT
      ii2.product_id,
      COUNT(DISTINCT ii2.invoice_id)::NUMERIC AS co_count
    FROM invoice_items ii1
    JOIN invoice_items ii2 ON ii2.invoice_id = ii1.invoice_id
      AND ii2.product_id != ALL(p_current_product_ids)
    JOIN invoices i ON i.id = ii1.invoice_id
    WHERE i.business_id = p_business_id
      AND ii1.product_id = ANY(p_current_product_ids)
      AND i.created_at >= NOW() - INTERVAL '90 days'
    GROUP BY ii2.product_id
    ORDER BY co_count DESC LIMIT p_limit * 2
  ),
  customer_history AS (
    -- Products this specific customer bought before
    SELECT DISTINCT ii.product_id
    FROM invoice_items ii
    JOIN invoices i ON i.id = ii.invoice_id
    WHERE i.contact_id = p_contact_id
      AND i.business_id = p_business_id
      AND ii.product_id != ALL(p_current_product_ids)
  )
  SELECT
    p.id, p.name, p.unit_price,
    CASE WHEN ch.product_id IS NOT NULL
         THEN 'Ramesh usually buys this too'
         ELSE 'Often bought together'
    END AS reason,
    ROUND(cp.co_count /
      NULLIF((SELECT COUNT(DISTINCT invoice_id) FROM invoice_items
        WHERE product_id = ANY(p_current_product_ids)), 0) * 100, 0) AS rate
  FROM co_purchased cp
  JOIN products p ON p.id = cp.product_id AND p.business_id = p_business_id
  LEFT JOIN customer_history ch ON ch.product_id = cp.product_id
  WHERE p.quantity > 0
  ORDER BY ch.product_id IS NOT NULL DESC, cp.co_count DESC
  LIMIT p_limit;
END; $$ LANGUAGE plpgsql;

export function useInvoiceMargin(lineItems: LineItem[]) {
  return useMemo(() => {
    const totalRevenue = lineItems.reduce((sum, item) =>
      sum + (item.quantity * item.unit_price), 0)
    const totalCost = lineItems.reduce((sum, item) =>
      sum + (item.quantity * (item.cost_price ?? 0)), 0)

      totalCost,
      grossProfit,
      marginPct,

  }, [lineItems])
}

CREATE OR REPLACE FUNCTION trg_update_invoice_on_payment()
RETURNS TRIGGER AS $$
DECLARE
  v_total_paid NUMERIC;
  v_invoice_total NUMERIC;
BEGIN
  SELECT SUM(amount) INTO v_total_paid
  FROM invoice_payments WHERE invoice_id = NEW.invoice_id;

  SELECT total_amount INTO v_invoice_total
  FROM invoices WHERE id = NEW.invoice_id;

  IF v_total_paid >= v_invoice_total THEN
    UPDATE invoices SET status = 'paid', updated_at = NOW()
    WHERE id = NEW.invoice_id;
  ELSE
    UPDATE invoices SET
      status = 'partial',
      partial_paid_amount = v_total_paid,
      partial_paid_at = COALESCE(partial_paid_at, NOW())
    WHERE id = NEW.invoice_id;
  END IF;

  RETURN NEW;
END; $$ LANGUAGE plpgsql;

## ─── SECTION G11: DATABASE HEALTH CHECK ─────────────────────────────────────

```
Paste this into Supabase SQL Editor to get a full database health report.
Run each query separately and share the results with Gemini for analysis.

-- QUERY 1: Tables missing RLS
SELECT tablename, 
  CASE WHEN rowsecurity THEN 'RLS ENABLED' ELSE '⚠️ RLS MISSING' END as rls_status
FROM pg_tables t
JOIN pg_class c ON c.relname = t.tablename
WHERE schemaname = 'public'
ORDER BY rowsecurity ASC;

-- QUERY 2: Tables missing indexes on business_id
SELECT t.tablename,
  CASE WHEN i.indexname IS NOT NULL 
       THEN 'Index exists' 
       ELSE '⚠️ Missing index on business_id' 
  END as index_status
FROM pg_tables t
LEFT JOIN pg_indexes i ON i.tablename = t.tablename 
  AND i.indexdef LIKE '%business_id%'
WHERE t.schemaname = 'public'
ORDER BY i.indexname ASC NULLS FIRST;

-- QUERY 3: Largest tables (performance risk)
SELECT relname AS table_name,
  pg_size_pretty(pg_total_relation_size(relid)) AS total_size,
  n_live_tup AS row_count
FROM pg_stat_user_tables
ORDER BY pg_total_relation_size(relid) DESC;

-- QUERY 4: Slow queries (run after enabling pg_stat_statements)
SELECT query, calls, 
  round(mean_exec_time::numeric, 2) AS avg_ms,
  round(total_exec_time::numeric, 2) AS total_ms
FROM pg_stat_statements
WHERE query NOT LIKE '%pg_%'
ORDER BY mean_exec_time DESC
LIMIT 10;

-- QUERY 5: Missing foreign key indexes
SELECT conname AS constraint_name,
  conrelid::regclass AS table_name,
  a.attname AS column_name
FROM pg_constraint c
JOIN pg_attribute a ON a.attrelid = c.conrelid 
  AND a.attnum = ANY(c.conkey)
WHERE c.contype = 'f'
AND NOT EXISTS (
  SELECT 1 FROM pg_index i
  WHERE i.indrelid = c.conrelid
  AND a.attnum = ANY(i.indkey)
)
ORDER BY table_name;

-- QUERY 6: Triggers on invoices table
SELECT trigger_name, event_manipulation, action_timing, action_statement
FROM information_schema.triggers
WHERE event_object_table = 'invoices'
ORDER BY trigger_name;

-- QUERY 7: Check for orphaned records
SELECT 'invoice_items with no parent invoice' as check_name,
  COUNT(*) as orphan_count
FROM invoice_items ii
LEFT JOIN invoices i ON i.id = ii.invoice_id
WHERE i.id IS NULL
UNION ALL
SELECT 'stock_movements with no product',
  COUNT(*)
FROM stock_movements sm
LEFT JOIN products p ON p.id = sm.product_id
WHERE p.id IS NULL;

Share the output with Gemini and say:

Using the system context from Section G0, analyse these results
and provide the exact SQL fixes for every issue found."
